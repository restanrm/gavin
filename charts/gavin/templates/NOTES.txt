1. Get the application URL by running these commands:
{{- if .Values.ingress.enabled }}
{{- range $host := .Values.ingress.hosts }}
  {{- range .paths }}
  http{{ if $.Values.ingress.tls }}s{{ end }}://{{ $host.host }}{{ .path }}
  {{- end }}
{{- end }}
{{- else if contains "NodePort" .Values.service.type }}
  export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "gavin.fullname" . }})
  export NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
  echo http://$NODE_IP:$NODE_PORT
{{- else if contains "LoadBalancer" .Values.service.type }}
     NOTE: It may take a few minutes for the LoadBalancer IP to be available.
           You can watch the status of by running 'kubectl get --namespace {{ .Release.Namespace }} svc -w {{ include "gavin.fullname" . }}'
  export SERVICE_IP=$(kubectl get svc --namespace {{ .Release.Namespace }} {{ include "gavin.fullname" . }} --template "{{"{{ range (index .status.loadBalancer.ingress 0) }}{{.}}{{ end }}"}}")
  echo http://$SERVICE_IP:{{ .Values.service.port }}
{{- else if contains "ClusterIP" .Values.service.type }}
  export POD_NAME=$(kubectl get pods --namespace {{ .Release.Namespace }} -l "app.kubernetes.io/name={{ include "gavin.name" . }},app.kubernetes.io/instance={{ .Release.Name }}" -o jsonpath="{.items[0].metadata.name}")
  export CONTAINER_PORT=$(kubectl get pod --namespace {{ .Release.Namespace }} $POD_NAME -o jsonpath="{.spec.containers[0].ports[0].containerPort}")
  echo "Visit http://127.0.0.1:8080 to use your application"
  kubectl --namespace {{ .Release.Namespace }} port-forward $POD_NAME 8080:$CONTAINER_PORT
{{- end }}

2. Database and uploads are persisted to:
{{- if .Values.persistence.enabled }}
  PVC: {{ include "gavin.pvcName" . }}
  Mount: {{ .Values.persistence.mountPath }}
{{- else }}
  WARNING: Persistence is disabled! Data will be lost on pod restart.
{{- end }}

3. OIDC Configuration:
{{- if .Values.existingSecret }}
  Using existing secret: {{ .Values.existingSecret }}
{{- else if .Values.secrets.oidcClientId }}
  Using secrets from values (not recommended for production!)
{{- else }}
  WARNING: No OIDC secrets configured! Application will fail to start.
  Please configure either:
    - existingSecret: name of a pre-created secret
    - secrets.* values in your values.yaml (not recommended)
{{- end }}

4. Pocket ID Callback URL Configuration:
  Your OIDC redirect URL should be configured in your Pocket ID (or OIDC provider) as:
{{- if .Values.ingress.enabled }}
  {{- range $host := .Values.ingress.hosts }}
  https://{{ $host.host }}/api/auth/callback
  {{- end }}
{{- else }}
  http://<your-domain>/api/auth/callback
{{- end }}

  This must match the OIDC_REDIRECT_URL environment variable.

5. Check application health:
  kubectl --namespace {{ .Release.Namespace }} get pods -l "app.kubernetes.io/name={{ include "gavin.name" . }}"
  kubectl --namespace {{ .Release.Namespace }} logs -f deployment/{{ include "gavin.fullname" . }}

6. Upgrade this release:
  helm upgrade {{ .Release.Name }} {{ .Chart.Name }} --namespace {{ .Release.Namespace }} -f your-values.yaml

For more information, visit: https://github.com/restanrm/gavin
